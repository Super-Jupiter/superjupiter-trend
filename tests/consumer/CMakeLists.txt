cmake_minimum_required(VERSION 3.16)
project(SuperJupiterTrendConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_TREND_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_TREND_SOURCE_DIR}" trend)
  if(TARGET superjupiter-trend-tests OR TARGET superjupiter-trend-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterTrend 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-trend-consumer main.cpp)
target_link_libraries(superjupiter-trend-consumer PRIVATE SuperJupiter::Trend)
